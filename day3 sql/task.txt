-- 16. 
SELECT d.name AS Dep_Name, d.gender AS Dep_Gender
FROM dependences d
JOIN Employee e ON d.Emp_num = e.ssn
WHERE d.gender = 'F' AND e.gender = 'F'
UNION
SELECT d.name AS Dep_Name, d.gender AS Dep_Gender
FROM dependences d
JOIN Employee e ON d.Emp_num = e.ssn
WHERE d.gender = 'M' AND e.gender = 'M';



-- 17. 
SELECT p.name, SUM(ep.hours) AS Total_Hours
FROM Project p
LEFT JOIN Emp_project ep ON p.Number = ep.pro_num
GROUP BY p.Number, p.name;


-- 18. 
SELECT d.*
FROM Department d
JOIN Employee e ON d.number = e.dep_num
WHERE e.ssn = (SELECT MIN(ssn) FROM Employee);



-- 19. 
SELECT d.name, MAX(e.Salary) AS Max_Salary, MIN(e.Salary) AS Min_Salary, AVG(e.Salary) AS Avg_Salary
FROM Department d
JOIN Employee e ON d.number = e.dep_num
GROUP BY d.number, d.name;

BEGIN
  UPDATE employee
  SET salary = salary * 1.1
  WHERE dep_num_fk = p_dep;
END$$

DELIMITER $$
CREATE PROCEDURE Total_Employees (IN p_dep INT, OUT p_count INT)
BEGIN
  SELECT COUNT(*) INTO p_count FROM employee WHERE dep_num_fk = p_dep;
END$$
DELIMITER ;

5. 
DELIMITER $$
CREATE PROCEDURE Highest_Paid_Per_Dept ()
BEGIN
  SELECT E.dep_num_fk, E.name, E.salary
  FROM employee E
  WHERE E.salary = (SELECT MAX(salary) FROM employee WHERE dep_num_fk = E.dep_num_fk);
END$$
DELIMITER ;

​When Indexes Improve Query Performance:
​Faster Searching (WHERE clause): They allow the database engine to quickly locate specific rows without scanning the entire table (avoiding full table scans).
​Optimized Joins (JOIN operations): They significantly speed up table join operations when foreign keys and primary keys are indexed.
​Efficient Sorting and Grouping (ORDER BY & GROUP BY): They eliminate the need for additional sorting algorithms because data within the index structure is already sorted.
​When Indexes May Reduce Query Performance:
​Slower Data Modification (INSERT, UPDATE, DELETE): Every time data is inserted, updated, or deleted, the database must also update the corresponding index structures, causing write performance overhead.
​Additional Storage Overhead: Indexes consume additional disk space and RAM, which can impact overall system memory management.
​Small Tables: For small tables, reading through the index can actually take longer than simply scanning the entire table directly.
